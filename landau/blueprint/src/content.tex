\chapter{Introduction}

This blueprint describes the formalization of the characterization of global steady states of the Vlasov-Maxwell-Landau (VML) system on a flat 3-torus $\T^3$. The main result (Theorem~42 in the notation of the original paper) states that any smooth, positive, integrable steady state of the VML system with sufficient velocity decay must be a spatially homogeneous equilibrium Maxwellian, with vanishing electric field and constant magnetic field.

\chapter{Definitions}\label{ch:defs}

\begin{definition}[Norm squared]\label{def:normSq}
  \lean{VML.normSq}
  \leanok
  For $v \in \R^3$, define $|v|^2 = \sum_{i=1}^{3} v_i^2$.
\end{definition}

\begin{definition}[Euclidean norm]\label{def:eucNorm}
  \lean{VML.eucNorm}
  \leanok
  \uses{def:normSq}
  For $v \in \R^3$, define $|v| = \sqrt{|v|^2}$.
\end{definition}

\begin{definition}[Landau matrix]\label{def:landauMatrix}
  \lean{VML.landauMatrix}
  \leanok
  \uses{def:eucNorm}
  Given a kernel function $\Psi : \R \to \R$, the Landau matrix $A(z)$ is the $3 \times 3$ matrix with entries
  \[
    A_{ij}(z) = \Psi(|z|)\bigl(\delta_{ij} |z|^2 - z_i z_j\bigr).
  \]
\end{definition}

\begin{definition}[Maxwellian]\label{def:isMaxwellian}
  \lean{VML.IsMaxwellian}
  \leanok
  \uses{def:normSq}
  A function $f : \R^3 \to \R$ is \emph{Maxwellian} if there exist $a_0 \in \R$, $b \in \R^3$, and $c_0 < 0$ such that
  \[
    f(v) = \exp\bigl(a_0 + b \cdot v + c_0 |v|^2\bigr)
  \]
  for all $v \in \R^3$.
\end{definition}

\begin{definition}[Equilibrium Maxwellian]\label{def:equilibriumMaxwellian}
  \lean{VML.equilibriumMaxwellian}
  \leanok
  \uses{def:normSq}
  For $\rho_{\mathrm{ion}} > 0$ and $T > 0$, the equilibrium Maxwellian is
  \[
    M_{\rho,T}(v) = \frac{\rho_{\mathrm{ion}}}{(2\pi T)^{3/2}} \exp\!\left(-\frac{|v|^2}{2T}\right).
  \]
\end{definition}

\begin{definition}[Landau collision operator]\label{def:landauOperator}
  \lean{VML.LandauOperator}
  \leanok
  \uses{def:landauMatrix}
  The Landau collision operator is
  \[
    Q(f, f)(v) = \grad_v \cdot \int_{\R^3} A(v - w)\bigl[f(w)\,\grad_v f(v) - f(v)\,\grad_w f(w)\bigr]\, dw.
  \]
\end{definition}

\begin{definition}[Entropy dissipation]\label{def:entropyDissipation}
  \lean{VML.entropyDissipation}
  \leanok
  \uses{def:landauOperator}
  The entropy dissipation functional is
  \[
    D(f) = \int_{\R^3} Q(f, f)(v)\, \log f(v)\, dv.
  \]
\end{definition}

\begin{definition}[Flat 3-torus (abstract)]\label{def:flatTorus3}
  \lean{VML.FlatTorus3}
  \leanok
  A \emph{flat 3-torus} is a compact, nonempty, first-countable measure space $X$ equipped with spatial gradient, divergence, and curl operators, a differentiability predicate \texttt{IsSpatiallyDiff}, and axioms encoding:
  operator linearity, closure of \texttt{IsSpatiallyDiff} under const/add/smul/log/grad,
  integration by parts, curl-integral vanishing, harmonic $\Rightarrow$ constant,
  maximum principle, Killing/curl-free/div-free $\Rightarrow$ harmonic, and Fubini for spatial--velocity double integrals.
\end{definition}

\begin{definition}[Velocity decay conditions]\label{def:velocityDecayConditions}
  \lean{VML.VelocityDecayConditions}
  \leanok
  \uses{def:flatTorus3}
  The \texttt{VelocityDecayConditions} structure bundles 17 integrability/Fubini/IBP conditions on $f$, $E$, $B$ that justify the analytical manipulations (integration by parts on $\R^3$, Fubini exchange, Leibniz differentiation under the integral, entropy dissipation continuity, etc.).
\end{definition}


\chapter{Landau Matrix Properties}\label{ch:section2}

\begin{lemma}[Positive semi-definiteness]\label{lem:landauMatrix_posSemidef}
  \lean{VML.landauMatrix_posSemidef}
  \leanok
  \uses{def:landauMatrix}
  If $\Psi(|z|) \ge 0$, then the Landau matrix $A(z)$ is positive semi-definite:
  $Y^T A(z) Y \ge 0$ for all $Y \in \R^3$.
\end{lemma}

\begin{lemma}[Nullspace characterization]\label{lem:landauMatrix_quadForm_eq_zero_iff}
  \lean{VML.landauMatrix_quadForm_eq_zero_iff}
  \leanok
  \uses{def:landauMatrix}
  If $\Psi(|z|) > 0$ and $z \ne 0$, then $Y^T A(z) Y = 0$ iff $Y \parallel z$.
\end{lemma}


\chapter{H-Theorem and Maxwellian Characterization}\label{ch:section3}

\begin{theorem}[H-theorem ($D(f) \le 0$)]\label{thm:H_theorem}
  \lean{VML.H_theorem}
  \leanok
  \uses{def:entropyDissipation, def:landauMatrix, def:landauOperator}
  If $\Psi \ge 0$ and $f > 0$ is smooth, then the entropy dissipation is non-positive: $D(f) \le 0$.
\end{theorem}

\begin{lemma}[$D = 0$ implies Maxwellian]\label{lem:D_zero_implies_maxwellian}
  \lean{VML.D_zero_implies_maxwellian}
  \leanok
  \uses{lem:landauMatrix_quadForm_eq_zero_iff, def:isMaxwellian, def:entropyDissipation}
  If $\Psi > 0$, $f > 0$, $f$ smooth and integrable, and $D(f) = 0$, then $f$ is Maxwellian.
  This is the core analytic step: $D = 0$ forces the PSD integrand to vanish, which by the nullspace characterization of $A(z)$ forces $\grad \log f$ to be affine, hence $f = \exp(\text{quadratic})$.
\end{lemma}

\begin{theorem}[Nullspace sufficiency (Maxwellian $\Rightarrow Q = 0$)]\label{thm:nullspace_sufficiency}
  \lean{VML.nullspace_sufficiency}
  \leanok
  \uses{def:landauMatrix, def:landauOperator}
  If $\log f(v) = a_0 + b \cdot v + c_0 |v|^2$ with $c_0 < 0$, then $Q(f,f)(v) = 0$ for all $v$.
\end{theorem}


\chapter{Transport Constraints}\label{ch:section4}

\begin{theorem}[Steady state is local Maxwellian]\label{thm:steady_state_is_local_maxwellian}
  \lean{VML.steady_state_is_local_maxwellian}
  \leanok
  \uses{lem:D_zero_implies_maxwellian, def:isMaxwellian, def:entropyDissipation}
  If $\Psi > 0$, $f > 0$, $f$ smooth, and $D(f(x, \cdot)) = 0$ for all $x$, then for every $x$, $f(x, \cdot)$ is Maxwellian.
\end{theorem}

\begin{lemma}[Lorentz force is divergence-free]\label{lem:lorentz_force_div_zero}
  \lean{VML.lorentz_force_div_zero}
  \leanok
  For any $E_0, B_0 \in \R^3$, $\divop_v (E_0 + v \times B_0) = 0$.
\end{lemma}

\begin{lemma}[Velocity-space IBP]\label{lem:velocity_ibp}
  \lean{VML.velocity_ibp}
  \leanok
  Integration by parts on $\R^3$ in velocity space: $\int \partial_i g \cdot h = -\int g \cdot \partial_i h$ under suitable decay.
\end{lemma}

\begin{lemma}[Force transport vanishes]\label{lem:force_transport_zero}
  \lean{VML.force_transport_zero}
  \leanok
  \uses{lem:velocity_ibp, lem:lorentz_force_div_zero}
  For any Lorentz force $(E_0 + v \times B_0)$, velocity-space IBP combined with $\divop_v(E_0 + v \times B_0) = 0$ gives
  $\int (E_0 + v \times B_0) \cdot \grad_v f \cdot \log f\, dv = 0$.
\end{lemma}

\begin{theorem}[Transport entropy identity ($\int_X D = 0$)]\label{thm:transport_entropy_from_vlasov}
  \lean{VML.transport_entropy_from_vlasov}
  \leanok
  \uses{lem:force_transport_zero, def:flatTorus3, def:entropyDissipation, def:landauOperator}
  Under the Vlasov equation, combined with spatial IBP (on the torus) and the force transport lemma:
  $\int_X D(f(x, \cdot))\, dx = 0$.
\end{theorem}

\chapter{Symmetrized Weak Form}\label{ch:swf}

\begin{lemma}[Landau IBP (symmetrized weak form)]\label{lem:landau_ibp}
  \lean{VML.landau_ibp}
  \leanok
  \uses{lem:velocity_ibp, def:landauMatrix, def:landauOperator}
  Integration by parts applied to $\int Q(f,f) \log f$, rewriting the collision operator in divergence form and integrating by parts in $v$:
  \[
    D(f) = -\int\int \ip{\grad_v \log f}{A(v-w)[f(w)\grad f(v) - f(v)\grad f(w)]}\, dw\, dv.
  \]
\end{lemma}

\begin{lemma}[Fubini symmetrization]\label{lem:fubini_symmetrization_logf}
  \lean{VML.fubini_symmetrization_logf}
  \leanok
  \uses{def:landauMatrix}
  The double integral from Landau IBP can be symmetrized by swapping $v \leftrightarrow w$ and using $A(v-w) = A(w-v)$, yielding the PSD integrand form with $(\grad \log f(v) - \grad \log f(w))$.
\end{lemma}

\begin{theorem}[Theorem 42 (abstract)]\label{thm:theorem42}
  \lean{VML.Theorem42}
  \leanok
  \uses{thm:main_from_physics, thm:H_theorem, thm:transport_entropy_from_vlasov, thm:polynomial_identity_from_vlasov, thm:poisson_boltzmann_from_vlasov, lem:landau_ibp, lem:fubini_symmetrization_logf, def:velocityDecayConditions, def:flatTorus3, def:equilibriumMaxwellian, def:entropyDissipation}
  Let $X$ be a flat 3-torus, $f > 0$ smooth and integrable, $\Psi > 0$.
  From the Vlasov equation + Maxwell equations + velocity decay conditions:
  \begin{enumerate}
    \item Landau IBP + Fubini symmetrization $\Rightarrow$ score form of $D(f)$
    \item $H$-theorem: $D(f(x,\cdot)) \le 0$ for each $x$
    \item Transport entropy identity: $\int_X D = 0$
    \item Combining: $D(f(x,\cdot)) = 0$ for all $x$
    \item Polynomial identity + Poisson--Boltzmann from the Vlasov equation
    \item Main from physics: conclude $f = M_{\rho,T}$, $E = 0$, $B = B_0$
  \end{enumerate}
\end{theorem}
